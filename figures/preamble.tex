% Shared TikZ setup for course diagrams.
%
% Loaded by two consumers:
%   1. figures/Makefile, which wraps each figure in a standalone document
%      and renders PNG/SVG for the website
%   2. the LaTeX lecture handouts in ../optimization-private/lecture-notes,
%      which \input the same figure sources directly
%
% Because both consumers share this file, a diagram looks identical on the
% website and in the printed course pack. Keep it free of anything
% document-class specific.

\usepackage{tikz}
\usepackage{amsmath}
\usepackage{amssymb}

\usetikzlibrary{arrows.meta, shapes, positioning, calc, patterns}

% Carried over from the sphinxcontrib.tikz configuration in the retired
% _config.yml, so diagrams authored against the old setup still compile.
\tikzset{
  block/.style = {draw, rectangle, minimum height=1cm, minimum width=2cm,
                  align=center, fill=blue!20},
  arrow/.style = {->, >=stealth, thick},
}
