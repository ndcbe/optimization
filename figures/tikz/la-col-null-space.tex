% Column space and null space of a rank-deficient matrix: the plane collapses
% onto a line, and a whole line of inputs is sent to zero.
%
%   figures/tikz/la-col-null-space.tex
%       -> media/figures/la-col-null-space.{png,svg}
%
% Lecture 12 (Linear Algebra Review), Rank section, under rank-nullity.
% Alex's decision D13.
%
% GEOMETRY (cm; each panel is a 4.4 x 4.4 window with the origin at its centre).
%   A = [0.5 1; 0.5 1], a 2 x 2 matrix of rank 1 (second column = 2 x first).
%   Range (column space): A x = (0.5 x1 + x2) (1, 1), so the range is the
%     line y = x, drawn from (-2,-2) to (2,2); dimension 1 = r(A).
%   Null space: A w = 0  <=>  0.5 w1 + w2 = 0, so w = t (2,-1); drawn from
%     (-2, 1) to (2,-1); dimension 1. Check: r(A) + dim N(A) = 1 + 1 = 2, the
%     number of columns.
%   Two inputs with the same image: x1 = (0.6, 0.8) and x2 = x1 + (1,-0.5) =
%     (1.6, 0.3), which differ by a null-space vector. A x1 = 0.5*0.6 + 0.8 =
%     1.1 and A x2 = 0.5*1.6 + 0.3 = 1.1, so BOTH land on the point (1.1, 1.1)
%     of the range line.
%
% GREYSCALE (scripts/check_greyscale.py). The null space is DASHED, the range is
% SOLID and thick; the two shared-image points are labelled. Hue is only a
% second cue.
\definecolor{okabeblue}{HTML}{0072B2}
\definecolor{okabevermillion}{HTML}{D55E00}

\begin{tikzpicture}[
    >={Latex[length=2.0mm]},
    font=\small,
    win/.style={draw=black!35, line width=0.4pt},
    ax/.style={draw=black!35, line width=0.4pt},
    rangeline/.style={draw=okabeblue, line width=2.2pt},
    nullline/.style={draw=okabevermillion, line width=1.4pt, dashed},
    pt/.style={fill=black, draw=black, circle, inner sep=0pt, minimum size=3.6pt},
    lab/.style={font=\small, inner sep=1.5pt},
    panelcap/.style={font=\small, align=center, text width=48mm, anchor=north},
  ]

% =====================================================================
% Domain: the null space is the line of inputs sent to zero
% =====================================================================
\begin{scope}[xshift=2.2cm, yshift=2.2cm]
  \draw[win] (-2.2,-2.2) rectangle (2.2,2.2);
  \draw[ax] (-2.2,0) -- (2.2,0);
  \draw[ax] (0,-2.2) -- (0,2.2);
  \draw[nullline] (-2.0,1.0) -- (2.0,-1.0);
  \node[lab, text=okabevermillion, anchor=south west, fill=white, inner sep=1pt] at (0.9,-1.6)
        {null space};
  % two inputs that differ by a null-space vector
  \node[pt] (x1) at (0.6,0.8) {};
  \node[pt] (x2) at (1.6,0.3) {};
  \draw[dotted, line width=0.8pt] (x1) -- (x2);
  \node[lab, anchor=south east] at (x1) {$\mathbf{x}_1$};
  \node[lab, anchor=south west] at (x2) {$\mathbf{x}_2$};
  \node[panelcap] at (0,-2.5) {\textbf{inputs} $\mathbb{R}^2$\\[-1pt]
        $\mathbf{x}_2 - \mathbf{x}_1$ is in the null space};
\end{scope}

% the map
\draw[->, line width=1.1pt] (4.7,2.2) -- (6.1,2.2) node[midway, above] {$\mathbf{A}$};

% =====================================================================
% Codomain: the range is the line the plane collapses onto
% =====================================================================
\begin{scope}[xshift=8.6cm, yshift=2.2cm]
  \draw[win] (-2.2,-2.2) rectangle (2.2,2.2);
  \draw[ax] (-2.2,0) -- (2.2,0);
  \draw[ax] (0,-2.2) -- (0,2.2);
  \draw[rangeline] (-2.0,-2.0) -- (2.0,2.0);
  \node[lab, text=okabeblue, anchor=north west, fill=white, inner sep=1pt] at (0.3,-0.5)
        {range};
  \node[pt] (y) at (1.1,1.1) {};
  \node[lab, anchor=south east, fill=white, inner sep=1pt] at (1.05,1.3) {$\mathbf{A}\mathbf{x}_1 = \mathbf{A}\mathbf{x}_2$};
  \node[panelcap] at (0,-2.5) {\textbf{outputs} $\mathbb{R}^2$\\[-1pt]
        range (column space) is a line};
\end{scope}

\node[font=\small, anchor=south] at (5.4,4.5)
  {$\mathbf{A} = \begin{bmatrix} 0.5 & 1 \\ 0.5 & 1 \end{bmatrix}$, \; $r(\mathbf{A}) = 1$};

\end{tikzpicture}
