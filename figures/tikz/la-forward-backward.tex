% The exact stored system vs the nearby system solved exactly by xhat.
% Backward error changes A only; the forward arrow compares solutions.
\begin{tikzpicture}[>=Latex,font=\small]
\node[draw,minimum width=3.4cm,minimum height=1cm,align=center,fill=blue!5] (a) at (0,2.5) {stored system\\$\mathbf A\mathbf x=\mathbf b$};
\node[draw,minimum width=4.5cm,minimum height=1cm,align=center,fill=blue!5] (b) at (7,2.5) {nearby system\\$(\mathbf A+\Delta\mathbf A)\hat{\mathbf x}=\mathbf b$};
\node (x) at (0,0) {exact solution $\mathbf x$};
\node (h) at (7,0) {computed solution $\hat{\mathbf x}$};
\draw[->,thick] (a)--(b) node[midway,above,align=center] {backward error\\$\min\|\Delta\mathbf A\|/\|\mathbf A\|$};
\draw[->,thick] (a)--(x) node[midway,left] {exact solve};
\draw[->,thick] (b)--(h) node[midway,right] {exact solve};
\draw[<->,thick,dashed] (x)--(h) node[midway,below,align=center] {forward error\\$\|\hat{\mathbf x}-\mathbf x\|/\|\mathbf x\|$};
\end{tikzpicture}
