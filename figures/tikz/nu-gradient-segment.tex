% Exact parameterized segment used in the Lipschitz remainder proof.
% Distances along the segment scale by t or 1-t; vector labels are bold.
\begin{tikzpicture}[x=1cm,y=1cm,font=\small,>=Latex]
\draw[very thick,blue!70!black] (0,0)--(9,0);
\fill (0,0) circle (2pt) node[below] {$\mathbf{x}^k$};
\fill (9,0) circle (2pt) node[below] {$\mathbf{x}^{k+1}$};
\fill[orange!80!black] (4.2,0) circle (2.5pt) node[below=5mm] {$\mathbf{x}^k+t\mathbf{p}^k$};
\draw[<->] (0,.6)--(4.2,.6) node[midway,above] {$t\|\mathbf{p}^k\|$};
\draw[<->,densely dashed] (0,1.35)--(9,1.35) node[midway,above] {$\|\mathbf{p}^k\|$};
\end{tikzpicture}
