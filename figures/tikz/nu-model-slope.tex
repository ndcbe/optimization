% Prepared scalar mechanism for differentiating a quadratic model.
% Illustration is m(p)=p^2-2p+3, m'(p)=2p-2, with stationary p=1.
\begin{tikzpicture}[x=1.5cm,y=.65cm,font=\small,>=Latex]
\begin{scope}
\draw[black,->] (-.35,0)--(2.45,0) node[right] {$p$};
\draw[black,->] (-.25,0)--(-.25,4.2) node[above] {$m(p)$};
\draw[very thick,blue!70!black,domain=-.25:2.25,samples=50] plot (\x,{\x*\x-2*\x+3});
\draw[densely dotted] (1,0)--(1,2);
\fill (1,2) circle (2pt);
\node[below] at (1,0) {$1$};
\node[above] at (1,4.8) {$m(p)=p^2-2p+3$};
\end{scope}
\begin{scope}[shift={(4.2,1.1)}]
\draw[black,->] (-.35,0)--(2.45,0) node[right] {$p$};
\draw[black,->] (-.25,-2.5)--(-.25,3) node[above] {$m'(p)$};
\draw[very thick,dashed,domain=-.2:2.2] plot (\x,{2*\x-2});
\fill (1,0) circle (2pt) node[below=2mm] {$1$};
\node[above] at (1,3.7) {$m'(p)=2p-2$};
\end{scope}
\end{tikzpicture}
