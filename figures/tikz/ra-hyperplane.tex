% A hyperplane {x : a'x = b}, its normal a, and the two half-spaces.
%
%   figures/tikz/ra-hyperplane.tex
%       -> media/figures/ra-hyperplane.{png,svg}
%
% Lecture 13 (Real Analysis Review), convexity.
%
% GEOMETRY (1 unit = 1 cm), n = 2. a = (3,4), unit normal (0.6,0.8). The line
% passes through x0 = (2.0,1.5), so b = a'x0 = 3*2 + 4*1.5 = 12. Direction of the
% line: d = (0.8,-0.6); a'd = 2.4 - 2.4 = 0, perpendicular by construction.
% Window [0,4.2] x [0,3.4]. The drawn normal is the UNIT normal scaled by 1.1 cm
% (a is only a direction here).  Point q = (3.4,2.6): a'q = 20.6 > 12, in the
% open half-space a'x > b. Point r = (0.6,0.6): a'r = 4.2 < 12, in a'x < b.
%
% GREYSCALE: the two sides are told apart by hatching and by their printed
% inequality, not by colour.
\usetikzlibrary{patterns}
\definecolor{okabeblue}{HTML}{0072B2}

\begin{tikzpicture}[
    >={Latex[length=2.0mm]},
    font=\small,
    lab/.style={font=\scriptsize, inner sep=1.5pt},
  ]
\def\wx{4.2}\def\wy{3.4}
\begin{scope}
  \clip (0,0) rectangle (\wx,\wy);
  % the half-space a'x <= b: hatched
  \path[pattern=north east lines, pattern color=okabeblue!50]
     (-2.8,5.1) -- (6.8,-2.1) -- (2.0,-8.5) -- (-7.6,-1.3) -- cycle;
  \draw[okabeblue, line width=1.3pt] (-2.8,5.1) -- (6.8,-2.1);
\end{scope}
\draw[black!40, line width=0.4pt] (0,0) rectangle (\wx,\wy);
% normal
\filldraw (2.0,1.5) circle (1.6pt);
\draw[->, line width=0.9pt] (2.0,1.5) -- (2.66,2.38);
\node[lab, anchor=south east] at (2.62,2.30) {$\mathbf{a}$};
\node[lab, anchor=north east] at (1.95,1.42) {$\mathbf{x}_0$};
% right-angle mark between the normal and the line direction (0.8,-0.6)
\draw[black!70, line width=0.4pt] (2.0,1.5) ++(0.2,0.267) -- ++(0.16,-0.12) -- ++(-0.2,-0.267);
% labels
\node[lab, anchor=south west, text=okabeblue, fill=white, inner sep=1pt] at (0.15,0.15)
     {$\{\mathbf{x}: \mathbf{a}^{\top}\mathbf{x} \le b\}$};
\node[lab, anchor=south east, fill=white, inner sep=1pt] at (4.1,2.9)
     {$\{\mathbf{x}: \mathbf{a}^{\top}\mathbf{x} \ge b\}$};
\node[lab, anchor=south west, rotate=-36.9] at (0.45,2.72)
     {$\mathbf{a}^{\top}\mathbf{x} = b$};
\node[font=\small, align=center, anchor=north] at (2.1,-0.15)
     {\textbf{hyperplane} $\mathbf{a}^{\top}\mathbf{x}=b$ and its two \textbf{half-spaces}};
\end{tikzpicture}
