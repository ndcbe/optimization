% Quasi-convex: every sublevel set {x : phi(x) <= alpha} is convex.
%
%   figures/tikz/ra-quasiconvex.tex
%       -> media/figures/ra-quasiconvex.{png,svg}
%
% Lecture 13 (Real Analysis Review), Convexity.
%
% GEOMETRY.
%  Left: phi(x) = 1 - exp(-x^2), alpha = 0.5. {phi <= 0.5} = [-s, s], with
%   s = sqrt(ln 2) = 0.8326: an interval, so phi is quasi-convex. It is not
%   convex: phi'' = (2 - 4x^2) exp(-x^2) < 0 for |x| > 1/sqrt 2 = 0.707.
%  Right: phi(x) = x^4 - 2x^2, alpha = -0.5. {phi <= alpha}: u = x^2 solves
%   u^2 - 2u + 0.5 <= 0, so u in [1 - 0.7071, 1 + 0.7071] = [0.2929, 1.7071],
%   i.e. |x| in [0.5412, 1.3066]: two intervals, not convex, so phi is not
%   quasi-convex (minima at +-1, phi = -1; phi(0) = 0).
%
% GREYSCALE: the sublevel set is a THICK solid bar on the level line (with a
% label); the level line is dashed; the graph is thin solid.
\definecolor{okabeblue}{HTML}{0072B2}
\definecolor{okabevermillion}{HTML}{D55E00}

\begin{tikzpicture}[
    >={Latex[length=2.0mm]},
    font=\small,
    lab/.style={font=\scriptsize, inner sep=1.5pt},
    panelcap/.style={font=\small, align=center, text width=62mm, anchor=north},
  ]
% ------------------------------------------------------------- left
\begin{scope}[x=1.3cm, y=2.4cm]
  \draw[->, black!55] (-2.2,0) -- (2.4,0);
  \draw[->, black!55] (0,-0.1) -- (0,1.15);
  \draw[very thick, domain=-2.2:2.2, smooth, samples=80, variable=\t]
    plot ({\t},{1-exp(-\t*\t)});
  \draw[dashed, black!70] (-2.2,0.5) -- (2.2,0.5);
  \node[lab, anchor=west] at (2.22,0.5) {$\alpha$};
  \draw[okabeblue, line width=2.4pt] (-0.8326,0.5) -- (0.8326,0.5);
  \draw[densely dotted] (-0.8326,0.5) -- (-0.8326,0);
  \draw[densely dotted] (0.8326,0.5) -- (0.8326,0);
    \node[lab, anchor=west] at (1.3,0.78) {$\phi = 1 - e^{-x^2}$};
  \node[panelcap] at (0,-0.2) {\textbf{quasi-convex, not convex}\\[-1pt] $\{\phi \le \alpha\}$ (thick) is one interval};
\end{scope}
% ------------------------------------------------------------- right
\begin{scope}[xshift=6.9cm, x=1.5cm, y=1.7cm]
  \draw[->, black!55] (-1.6,0) -- (1.8,0);
  \draw[->, black!55] (0,-1.1) -- (0,0.75);
  \draw[very thick, domain=-1.5:1.5, smooth, samples=80, variable=\t]
    plot ({\t},{\t*\t*\t*\t-2*\t*\t});
  \draw[dashed, black!70] (-1.6,-0.5) -- (1.6,-0.5);
  \node[lab, anchor=west] at (1.62,-0.5) {$\alpha$};
  \draw[okabevermillion, line width=2.4pt] (-1.3066,-0.5) -- (-0.5412,-0.5);
  \draw[okabevermillion, line width=2.4pt] (0.5412,-0.5) -- (1.3066,-0.5);
    \node[lab, anchor=west] at (0.15,0.6) {$\phi = x^4 - 2x^2$};
  \node[panelcap] at (0,-1.2) {\textbf{not quasi-convex}\\[-1pt] $\{\phi \le \alpha\}$ (thick) is two intervals};
\end{scope}
\end{tikzpicture}
