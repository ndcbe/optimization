% Strong convexity: a quadratic lower bound under the tangent line.
%
%   figures/tikz/ra-strong-convexity.tex
%       -> media/figures/ra-strong-convexity.{png,svg}
%
% Lecture 13 (Real Analysis Review), Convexity. m-strongly convex means
%   phi(x+p) >= phi(x) + grad phi(x)'p + (m/2)||p||^2   for all x, p
% (Nesterov 2018, Def. 2.1.3, p. 74). Convex only asks for the tangent line.
%
% GEOMETRY.
%  Left: phi(x) = x^2 + x^4/4, phi'' = 2 + 3x^2 >= 2, so m = 2. Expansion point
%   x = 0.6: phi = 0.3924, phi' = 1.416. Tangent T(t) = 0.3924 + 1.416 (t - 0.6);
%   quadratic bound Q(t) = T(t) + (t - 0.6)^2 (m/2 = 1). phi - Q equals the
%   Bregman gap of x^4/4, which is >= 0 (checked numerically, min = 0 at t = 0.6).
%   At t = -1.2: T = -2.157, Q = 1.084, phi = 1.958.
%  Right: phi(x) = x^4 at x = 0: tangent is the axis, bound Q(t) = (1/2) t^2
%   (m = 1). phi < Q for |t| < 1/sqrt(2) = 0.707, so the bound fails at every
%   m > 0 close to 0 (phi'' (0) = 0): x^4 is strictly convex, not strongly convex.
%
% GREYSCALE: graph solid, tangent dashed, bound dash-dotted; all directly labelled.
\definecolor{okabeblue}{HTML}{0072B2}
\definecolor{okabevermillion}{HTML}{D55E00}

\begin{tikzpicture}[
    >={Latex[length=2.0mm]},
    font=\small,
    lab/.style={font=\scriptsize, inner sep=1.5pt},
    panelcap/.style={font=\small, align=center, text width=62mm, anchor=north},
  ]
% ------------------------------------------------------------- left
\begin{scope}[x=1.5cm, y=0.7cm]
  \draw[->, black!55] (-1.5,0) -- (1.6,0);
  \draw[->, black!55] (0,-0.5) -- (0,4.4);
  \begin{scope}
    \clip (-1.5,-1.4) rectangle (1.6,4.6);
    \draw[dashed, okabeblue, thick, domain=-1.5:1.6, variable=\t]
      plot ({\t},{0.3924+1.416*(\t-0.6)});
    \draw[dash pattern=on 4pt off 1.5pt on 1pt off 1.5pt, thick, domain=-1.5:1.6, smooth, samples=50, variable=\t]
      plot ({\t},{0.3924+1.416*(\t-0.6)+(\t-0.6)*(\t-0.6)});
    \draw[very thick, domain=-1.5:1.5, smooth, samples=60, variable=\t]
      plot ({\t},{\t*\t+\t*\t*\t*\t/4});
  \end{scope}
  \filldraw (0.6,0.3924) circle (1.6pt);
  \node[lab, below right] at (0.6,0.3924) {$\mathbf{x}$};
  \node[lab, okabeblue, anchor=west] at (0.9,0.55) {tangent line};
  \node[lab, anchor=south east] at (-1.1,3.0) {$\phi$};
  \node[lab, anchor=west] at (-1.5,0.9) {bound};
  \node[panelcap] at (0.05,-1.5) {\textbf{strongly convex}\\[-1pt] $\phi \ge$ tangent $+\ \tfrac{m}{2}\lVert\mathbf{p}\rVert^2$};
\end{scope}
% ------------------------------------------------------------- right
\begin{scope}[xshift=7.0cm, x=1.5cm, y=1.1cm]
  \draw[->, black!55] (-1.4,0) -- (1.5,0);
  \draw[->, black!55] (0,-0.3) -- (0,2.3);
  \begin{scope}
    \clip (-1.4,-0.3) rectangle (1.5,2.3);
    \draw[dash pattern=on 4pt off 1.5pt on 1pt off 1.5pt, thick, domain=-1.4:1.4, smooth, samples=50, variable=\t]
      plot ({\t},{0.5*\t*\t});
    \draw[very thick, domain=-1.3:1.3, smooth, samples=60, variable=\t]
      plot ({\t},{\t*\t*\t*\t});
  \end{scope}
  \filldraw (0,0) circle (1.6pt);
  % phi lies below the parabola for |t| < 0.707
  \draw[okabevermillion, line width=1.4pt] (-0.707,-0.12) -- (0.707,-0.12);
  \node[lab, okabevermillion, anchor=north] at (0,-0.12) {$\phi$ below the parabola};
  \node[lab, anchor=west] at (0.15,1.95) {$\phi = x^4$};
  \node[lab, anchor=south west] at (1.0,0.9) {bound};
  \node[panelcap] at (0,-0.6) {\textbf{only strictly convex}\\[-1pt] at $x = 0$ the bound fails for every $m>0$};
\end{scope}
\end{tikzpicture}
