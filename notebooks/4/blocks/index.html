Blocks in Pyomo¶

Reference: Chapter 8: Structured Modeling with Blocks, Pyomo -- Optimization Modeling in Python, Bynum et al. (2021).

Blocks provide a convenient way to express hierarchically-structured models in Pyomo. As an example, consider the special structure in the Farmer’s example for Stochastic Programming.

Asking ChatGPT to rewrite the scenario-indexed version of our model from Stochastic Programming with blocks gives the following code (which was then slightly modified):

# Pyomo's unit library has acres and metric tons but no money, so declare one.
u.load_definitions_from_strings(["USD = [currency]"])

# Shorthands for the three compound units this problem uses
USD_PER_ACRE = u.USD / u.acre
USD_PER_TON = u.USD / u.metric_ton
TON_PER_ACRE = u.metric_ton / u.acre


def build_sp_model(yields):
    """
    Rewritten version of the stochastic programming model using blocks.

    Arguments:
        yields: Yield information as a list, following the rank [wheat, corn, beets]

    Return:
        model: farmer problem model using blocks
    """

    # Model
    model = ConcreteModel()

    # Sets
    all_crops = ["WHEAT", "CORN", "BEETS"]
    purchase_crops = ["WHEAT", "CORN"]
    sell_crops = ["WHEAT", "CORN", "BEETS_FAVORABLE", "BEETS_UNFAVORABLE"]
    scenarios = ["ABOVE", "AVERAGE", "BELOW"]

    # First-stage decision: how much to land to dedicate to each crop [acre]
    model.X = Var(all_crops, domain=NonNegativeReals, units=u.acre)

    # Define a block for each scenario
    def scenario_block_rule(b, scenario):
        # Second-stage decision variables [t]
        b.Y = Var(
            purchase_crops, domain=NonNegativeReals, units=u.metric_ton
        )  # How much to purchase in this scenario
        b.W = Var(
            sell_crops, domain=NonNegativeReals, units=u.metric_ton
        )  # How much to sell in this scenario

        # Purchase cost and sales revenue for this scenario [USD]
        if scenario == "ABOVE":
            b.purchase_cost = (
                238 * USD_PER_TON * b.Y["WHEAT"] + 210 * USD_PER_TON * b.Y["CORN"]
            )
            b.sales_revenue = (
                170 * USD_PER_TON * b.W["WHEAT"]
                + 150 * USD_PER_TON * b.W["CORN"]
                + 36 * USD_PER_TON * b.W["BEETS_FAVORABLE"]
                + 10 * USD_PER_TON * b.W["BEETS_UNFAVORABLE"]
            )
        elif scenario == "AVERAGE":
            b.purchase_cost = (
                238 * USD_PER_TON * b.Y["WHEAT"] + 210 * USD_PER_TON * b.Y["CORN"]
            )
            b.sales_revenue = (
                170 * USD_PER_TON * b.W["WHEAT"]
                + 150 * USD_PER_TON * b.W["CORN"]
                + 36 * USD_PER_TON * b.W["BEETS_FAVORABLE"]
                + 10 * USD_PER_TON * b.W["BEETS_UNFAVORABLE"]
            )
        else:  # BELOW
            b.purchase_cost = (
                238 * USD_PER_TON * b.Y["WHEAT"] + 210 * USD_PER_TON * b.Y["CORN"]
            )
            b.sales_revenue = (
                170 * USD_PER_TON * b.W["WHEAT"]
                + 150 * USD_PER_TON * b.W["CORN"]
                + 36 * USD_PER_TON * b.W["BEETS_FAVORABLE"]
                + 10 * USD_PER_TON * b.W["BEETS_UNFAVORABLE"]
            )

        # Scenario constraints [t]
        if scenario == "ABOVE":
            b.wheat_constraint = Constraint(
                expr=yields[0] * 1.2 * TON_PER_ACRE * model.X["WHEAT"]
                + b.Y["WHEAT"]
                - b.W["WHEAT"]
                >= 200 * u.metric_ton
            )
            b.corn_constraint = Constraint(
                expr=yields[1] * 1.2 * TON_PER_ACRE * model.X["CORN"]
                + b.Y["CORN"]
                - b.W["CORN"]
                >= 240 * u.metric_ton
            )
            b.beets_constraint = Constraint(
                expr=yields[2] * 1.2 * TON_PER_ACRE * model.X["BEETS"]
                - b.W["BEETS_FAVORABLE"]
                - b.W["BEETS_UNFAVORABLE"]
                >= 0 * u.metric_ton
            )
        elif scenario == "AVERAGE":
            b.wheat_constraint = Constraint(
                expr=yields[0] * TON_PER_ACRE * model.X["WHEAT"]
                + b.Y["WHEAT"]
                - b.W["WHEAT"]
                >= 200 * u.metric_ton
            )
            b.corn_constraint = Constraint(
                expr=yields[1] * TON_PER_ACRE * model.X["CORN"]
                + b.Y["CORN"]
                - b.W["CORN"]
                >= 240 * u.metric_ton
            )
            b.beets_constraint = Constraint(
                expr=yields[2] * TON_PER_ACRE * model.X["BEETS"]
                - b.W["BEETS_FAVORABLE"]
                - b.W["BEETS_UNFAVORABLE"]
                >= 0 * u.metric_ton
            )
        else:  # BELOW
            b.wheat_constraint = Constraint(
                expr=yields[0] * 0.8 * TON_PER_ACRE * model.X["WHEAT"]
                + b.Y["WHEAT"]
                - b.W["WHEAT"]
                >= 200 * u.metric_ton
            )
            b.corn_constraint = Constraint(
                expr=yields[1] * 0.8 * TON_PER_ACRE * model.X["CORN"]
                + b.Y["CORN"]
                - b.W["CORN"]
                >= 240 * u.metric_ton
            )
            b.beets_constraint = Constraint(
                expr=yields[2] * 0.8 * TON_PER_ACRE * model.X["BEETS"]
                - b.W["BEETS_FAVORABLE"]
                - b.W["BEETS_UNFAVORABLE"]
                >= 0 * u.metric_ton
            )

        # Set upper bounds for BEETS_FAVORABLE [t]
        b.W["BEETS_FAVORABLE"].setub(6000)

    # Create blocks for each scenario
    model.scenarios = Block(scenarios, rule=scenario_block_rule)

    # Objective function [USD]
    def objective_rule(m):
        planting_cost = (
            150 * USD_PER_ACRE * m.X["WHEAT"]
            + 230 * USD_PER_ACRE * m.X["CORN"]
            + 260 * USD_PER_ACRE * m.X["BEETS"]
        )
        expected_purchase_cost = (1 / 3) * sum(
            m.scenarios[sc].purchase_cost for sc in scenarios
        )
        expected_sales_revenue = (1 / 3) * sum(
            m.scenarios[sc].sales_revenue for sc in scenarios
        )
        return planting_cost + expected_purchase_cost - expected_sales_revenue

    model.OBJ = Objective(rule=objective_rule, sense=minimize)

    # First-stage constraint: total area allocated to crops should not exceed 500 [acre]
    model.total_land_constraint = Constraint(expr=summation(model.X) <= 500 * u.acre)

    # Raises UnitsError if any constraint or the objective is inconsistent
    assert_units_consistent(model)

    return model

Separate the Data from the Model¶

The above blocks example is not really a big improvement from our alternate implementation in Stochastic Programming.

In the above code (from ChatGPT), we see:

  • Data for each scenario are hardcoded into the model

  • If statements are used to toggle the correct constraints for each scenario

The above code is not general purpose; updating it to use alternate scenario data or consider additional crops would take a lot of manual effort. It would be easy to make mistakes.

Thus, it is best practice to always separate your specific problem data from the general mathematical model. Let’s see an example.

Single Scenario Optimization Problem¶

We dropped the Units row from the CSV above, so the units now exist only in our heads. Put them back where the solver can see them: every Param and Var below carries units=, and the builder ends with assert_units_consistent. The Yield column is the one that matters most --- t/acre is what converts a planting decision in acres into a production in tons, and nothing else in the code says so.

from pyomo.environ import Set, Param, Constraint, maximize


def build_deterministic_model(nominal_data, m=None, skip_objective=False):

    # Create concrete model (if not provided)
    if m is None:
        m = ConcreteModel()

    # Sets
    crops = nominal_data.index.to_list()
    m.CROPS = Set(initialize=crops)

    # Total land available [acre]
    m.max_area = 500 * u.acre

    # Parameters. The units are the ones the CSV declared and we dropped above.
    m.cost = Param(
        m.CROPS,
        initialize=nominal_data["Planting Cost"],
        domain=NonNegativeReals,
        units=USD_PER_ACRE,
    )
    m.sell_price_favorable = Param(
        m.CROPS,
        initialize=nominal_data["Favorable Selling Price"],
        domain=NonNegativeReals,
        units=USD_PER_TON,
    )
    m.sell_price_unfavorable = Param(
        m.CROPS,
        initialize=nominal_data["Unfavorable Selling Price"],
        domain=NonNegativeReals,
        units=USD_PER_TON,
    )
    m.purchase_price = Param(
        m.CROPS,
        initialize=nominal_data["Purchase Price"],
        domain=NonNegativeReals,
        units=USD_PER_TON,
    )
    m.crop_yield = Param(
        m.CROPS,
        initialize=nominal_data["Yield"],
        domain=NonNegativeReals,
        units=TON_PER_ACRE,
    )
    m.min_required = Param(
        m.CROPS,
        initialize=nominal_data["Minimum Requirement"],
        domain=NonNegativeReals,
        units=u.metric_ton,
    )
    m.max_possible = Param(
        m.CROPS,
        initialize=nominal_data["Maximum Favorable Production"],
        domain=NonNegativeReals,
        units=u.metric_ton,
    )

    # Extract boolean parameters
    m.enforce_max_production = nominal_data["Enforce Max Production"].to_dict()
    m.enforce_min_requirement = nominal_data["Enforce Min Requirement"].to_dict()
    m.allow_purchases = nominal_data["Allow Purchases"].to_dict()

    # Stage 1 decision variables [acre]
    m.plant = Var(m.CROPS, domain=NonNegativeReals, units=u.acre)

    # Stage 2 decision variables [t]
    m.buy = Var(m.CROPS, domain=NonNegativeReals, units=u.metric_ton)  # purchases
    m.sell_favorable = Var(
        m.CROPS, domain=NonNegativeReals, units=u.metric_ton
    )  # sales
    m.sell_unfavorable = Var(
        m.CROPS, domain=NonNegativeReals, units=u.metric_ton
    )  # sales

    for c in crops:
        # Disable purchases for crops with NaN prices
        if not m.allow_purchases[c]:
            m.buy[c].fix(0)

        # Enforce maximum production limits
        if m.enforce_max_production[c]:
            m.sell_favorable[c].setub(m.max_possible[c])

    # Total area constraint [acre]
    @m.Constraint()
    def total_area(m):
        return sum(m.plant[crop] for crop in m.CROPS) <= m.max_area

    # Constraint on the production of each crop [t]
    @m.Constraint(m.CROPS)
    def crop_min_constraint(m, crop):
        if m.enforce_min_requirement[crop]:
            return (
                m.plant[crop] * m.crop_yield[crop]
                + m.buy[crop]
                - m.sell_unfavorable[crop]
                - m.sell_favorable[crop]
                >= m.min_required[crop]
            )
        else:
            return (
                m.plant[crop] * m.crop_yield[crop]
                + m.buy[crop]
                - m.sell_unfavorable[crop]
                - m.sell_favorable[crop]
                >= 0 * u.metric_ton
            )

    # Maximize net profit [USD]
    @m.Expression()
    def net_profit(m):
        return (
            -sum(m.plant[c] * m.cost[c] for c in crops)
            - sum(m.buy[c] * m.purchase_price[c] for c in crops)
            + sum(
                m.sell_favorable[c] * m.sell_price_favorable[c]
                + m.sell_unfavorable[c] * m.sell_price_unfavorable[c]
                for c in crops
            )
        )

    if not skip_objective:
        m.obj = Objective(expr=m.net_profit, sense=maximize)

    # Raises UnitsError if any constraint or the objective is inconsistent
    assert_units_consistent(m)

    return m

def create_two_stage_stochastic_model(scenarios):

    # Model
    m = ConcreteModel()

    nominal_scenario_name = "AVERAGE"
    nominal_data = scenarios[nominal_scenario_name]

    # Sets
    crops = nominal_data.index.to_list()
    m.CROPS = Set(initialize=crops)

    # Total land available [acre]
    m.max_area = 500 * u.acre

    # Stage 1 variables [acre]
    m.plant = Var(
        m.CROPS,
        domain=NonNegativeReals,
        initialize=100,
        bounds=(0, m.max_area),
        units=u.acre,
    )

    # Stage 1 constraint [acre]
    @m.Constraint()
    def total_area(m):
        return sum(m.plant[crop] for crop in m.CROPS) <= m.max_area

    def second_stage_block(b, scenario_name):

        # print("name = ", scenario_name)

        # Grab data for this specific scenario
        s_data = scenarios[scenario_name]

        # Parameters. The units are the ones the CSV declared and we dropped above.
        b.cost = Param(
            m.CROPS,
            initialize=s_data["Planting Cost"],
            domain=NonNegativeReals,
            units=USD_PER_ACRE,
        )
        b.sell_price_favorable = Param(
            m.CROPS,
            initialize=s_data["Favorable Selling Price"],
            domain=NonNegativeReals,
            units=USD_PER_TON,
        )
        b.sell_price_unfavorable = Param(
            m.CROPS,
            initialize=s_data["Unfavorable Selling Price"],
            domain=NonNegativeReals,
            units=USD_PER_TON,
        )
        b.purchase_price = Param(
            m.CROPS,
            initialize=s_data["Purchase Price"],
            domain=NonNegativeReals,
            units=USD_PER_TON,
        )
        b.crop_yield = Param(
            m.CROPS,
            initialize=s_data["Yield"],
            domain=NonNegativeReals,
            units=TON_PER_ACRE,
        )
        b.min_required = Param(
            m.CROPS,
            initialize=s_data["Minimum Requirement"],
            domain=NonNegativeReals,
            units=u.metric_ton,
        )
        b.max_possible = Param(
            m.CROPS,
            initialize=s_data["Maximum Favorable Production"],
            domain=NonNegativeReals,
            units=u.metric_ton,
        )

        # Extract boolean parameters
        b.enforce_max_production = s_data["Enforce Max Production"].to_dict()
        b.enforce_min_requirement = s_data["Enforce Min Requirement"].to_dict()
        b.allow_purchases = s_data["Allow Purchases"].to_dict()

        # Stage 2 decision variables [t]
        b.buy = Var(
            m.CROPS, domain=NonNegativeReals, initialize=0, units=u.metric_ton
        )  # purchases
        b.sell_favorable = Var(
            m.CROPS, domain=NonNegativeReals, initialize=0, units=u.metric_ton
        )  # sales
        b.sell_unfavorable = Var(
            m.CROPS, domain=NonNegativeReals, initialize=0, units=u.metric_ton
        )  # sales

        for c in crops:
            # Disable purchases for crops with NaN prices
            if not b.allow_purchases[c]:
                b.buy[c].fix(0)

            # Enforce maximum production limits
            if b.enforce_max_production[c]:
                b.sell_favorable[c].setub(b.max_possible[c])

        # Constraint on the production of each crop [t]
        @b.Constraint(m.CROPS)
        def crop_min_constraint(b, crop):
            if b.enforce_min_requirement[crop]:
                return (
                    m.plant[crop] * b.crop_yield[crop]
                    + b.buy[crop]
                    - b.sell_unfavorable[crop]
                    - b.sell_favorable[crop]
                    >= b.min_required[crop]
                )
            else:
                return (
                    m.plant[crop] * b.crop_yield[crop]
                    + b.buy[crop]
                    - b.sell_unfavorable[crop]
                    - b.sell_favorable[crop]
                    >= 0 * u.metric_ton
                )

        # Net profit in this scenario [USD]
        @b.Expression()
        def scenario_net_profit(b):
            return (
                -sum(m.plant[c] * b.cost[c] for c in crops)
                - sum(b.buy[c] * b.purchase_price[c] for c in crops)
                + sum(
                    b.sell_favorable[c] * b.sell_price_favorable[c]
                    + b.sell_unfavorable[c] * b.sell_price_unfavorable[c]
                    for c in crops
                )
            )

    # Create blocks for each scenario
    m.planning_scenarios = Block(scenarios.keys(), rule=second_stage_block)

    # Maximize net profot [USD]
    @m.Objective(sense=maximize)
    def obj(m):
        return sum(
            m.planning_scenarios[sc].scenario_net_profit for sc in scenarios.keys()
        ) / len(scenarios)

    # Raises UnitsError if any constraint or the objective is inconsistent
    assert_units_consistent(m)

    return m

def create_two_stage_stochastic_model_better(scenarios):

    # Create a concrete model
    m = ConcreteModel()

    # Create a block for each scenario
    # This creates a copy of your deterministic model for each scenario
    m.planning_scenarios = Block(
        scenarios.keys(),
        rule=lambda b, s: build_deterministic_model(
            scenarios[s], m=b, skip_objective=True
        ),
    )

    # Enforce the same planting (stage 1) decisions for all scenarios [acre]
    m.SCENARIOS = Set(initialize=scenarios.keys())
    m.CROPS = Set(initialize=scenarios["AVERAGE"].index.to_list())

    @m.Constraint(m.SCENARIOS, m.CROPS)
    def enforce_same_planting(m, s, c):
        if s == "AVERAGE":
            return Constraint.Skip
        else:
            return (
                m.planning_scenarios[s].plant[c]
                == m.planning_scenarios["AVERAGE"].plant[c]
            )

    # Expected net profit [USD]
    @m.Objective(sense=maximize)
    def obj(m):
        return sum(m.planning_scenarios[s].net_profit for s in scenarios.keys()) / len(
            scenarios
        )

    # Raises UnitsError if any constraint or the objective is inconsistent
    assert_units_consistent(m)

    return m
