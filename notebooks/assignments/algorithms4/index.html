4d. What does a solver see?¶

The code below solves Case B from three starting points and Case C from one, reporting the termination condition, the solution, and the multipliers in the Lecture 19 sign convention. Run it. Solver versions differ, so report the values your run produces.

def solve_case(case, x_init):
    m = pyo.ConcreteModel()
    m.x1 = pyo.Var(initialize=x_init[0])
    m.x2 = pyo.Var(initialize=x_init[1])
    m.dual = pyo.Suffix(direction=pyo.Suffix.IMPORT)
    if case == "B":
        m.obj = pyo.Objective(expr=-m.x1 - 2 * m.x2)
        m.g1 = pyo.Constraint(expr=m.x1**2 + m.x2**2 - 5 <= 0)
        m.g2 = pyo.Constraint(expr=m.x1 + 2 * m.x2 - 5 <= 0)
    else:
        m.obj = pyo.Objective(expr=m.x1 + m.x2)
        m.g1 = pyo.Constraint(expr=(m.x1 - 1) ** 2 + m.x2**2 - 1 <= 0)
        m.g2 = pyo.Constraint(expr=(m.x1 + 1) ** 2 + m.x2**2 - 1 <= 0)
    results = pyo.SolverFactory("ipopt").solve(m)
    u = (-m.dual[m.g1], -m.dual[m.g2])  # Lecture 19 sign convention
    print(f"Case {case}, start {x_init}: {results.solver.termination_condition}")
    print(f"   x = ({pyo.value(m.x1):.6g}, {pyo.value(m.x2):.6g}),  f = {pyo.value(m.obj):.6g}")
    print(f"   u = ({u[0]:.6g}, {u[1]:.6g})")
    if case == "B":
        print(f"   2*u1 + u2 = {2 * u[0] + u[1]:.6g}")
    print(f"   g = ({pyo.value(m.g1.body):.3g}, {pyo.value(m.g2.body):.3g})")


for start in [(0, 0), (-1, 0), (3, 3)]:
